--- get_remote_price.js ---
(function($,Drupal){function alert_price(){var product_ids=[];$('div.display-price').each(function(){var class_list=$(this).attr('class').split(' ');if(class_list[2]!=undefined){product_ids.push(class_list[2].replace("uc-product-",""));}});var dataset={product_ids:product_ids};if(product_ids.length>0){var get_price_path=Drupal.settings.get_price_url;$.getJSON(get_price_path,dataset,function(resp){if(resp.price_processed){$.each(resp.product_price_list,function(key,value){$('div.uc-product-'+key).next('.sell-price').addClass('updated-price').html('Your Price: $'+value);});}
else{$('.sell-price').html('Failed to load the price');}
if(resp.rrp_price_processed){$.each(resp.product_rrp_price_list,function(key,value){$('div.uc-product-'+key).addClass('updated-price').find('.uc-price').html(value);});}
else{$('.display-price').html('Failed to load RRP the price');}});}}
Drupal.behaviors.get_remote_price={attach:function(context,settings){$('.block-latest-bought-products').once('mode_once',function(){alert_price();});$('body.uc-product-node').once('mode_once',function(){alert_price();});$('.view-search-products').once('mode_once',function(){alert_price();});$('.view-brands-landing-page').once('mode_once',function(){alert_price();});$('.view-colorbox-popup').once('mode_once',function(){alert_price();});$('.vehicle-search-colorbox-popup').once('mode_once',function(){alert_price();});var product_ids=[];$('#colorbox').find('.display-price').each(function(){$('#colorbox').find('form').attr('action',window.location.pathname);var class_list=$(this).attr('class').split(' ');if(class_list[2]!=undefined){product_ids.push(class_list[2].replace("uc-product-",""));}});var dataset={product_ids:product_ids};if(product_ids.length>0){var get_price_path=Drupal.settings.get_price_url;$.getJSON(get_price_path,dataset,function(resp){if(resp.processed){$.each(resp.product_price_list,function(key,value){$('#colorbox').find('.sell-price').addClass('updated-price').html('Your Price: $'+value)});}});}}};}(jQuery,Drupal));
